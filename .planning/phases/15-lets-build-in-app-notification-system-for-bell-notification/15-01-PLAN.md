  <name>Task 1: Create Notification Schema</name>
  <files>backend/src/database/schema/notifications.ts</files>
  <action>
    Create `backend/src/database/schema/notifications.ts` with Drizzle ORM table definition:

    ```typescript
    import { pgTable, text, timestamp, uuid, boolean, jsonb } from 'drizzle-orm/pg-core';
    import { organisations, users } from './index';

    export const notifications = pgTable('notifications', {
      id: uuid('id').defaultRandom().primaryKey(),
      organisationId: uuid('organisation_id')
        .notNull()
        .references(() => organisations.id, { onDelete: 'cascade' }),
      userId: uuid('user_id')
        .notNull()
        .references(() => users.id, { onDelete: 'cascade' }),
      titleKey: text('title_key').notNull(), // e.g., "quote.assigned"
      data: jsonb('data').default({}), // Template variables as JSON
      isRead: boolean('is_read').default(false),
      createdAt: timestamp('created_at').defaultNow(),
      expiresAt: timestamp('expires_at').defaultNow(), // Auto-set to NOW + 30 days
    });
    ```

    - organisationId + userId for org-scoped queries and user targeting
    - titleKey for template-based content (D-05)
    - data as jsonb for flexible metadata storage
    - isRead for read/unread state (D-03)
    - expiresAt for auto-expiry at 30 days (D-03)
    - CASCADE delete when org or user is deleted
  </action>
