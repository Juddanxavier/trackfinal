  <action>
    Create `backend/src/modules/notifications/notifications.service.ts` with:

    ```typescript
    import { Injectable } from '@nestjs/common';
    import { eq, and, desc } from 'drizzle-orm';
    import { notifications } from '../../database/schema';
    import { CreateNotificationDto, QueryNotificationsDto } from './dto';
    import { EventsGateway } from '../events/events.gateway';
    import { DrizzleService } from '../../database/drizzle.service';

    @Injectable()
    export class NotificationsService {
      constructor(
        private drizzle: DrizzleService,
        private eventsGateway: EventsGateway,
      ) {}

      async create(organisationId: string, dto: CreateNotificationDto) {
        // Auto-set expiresAt to now + 30 days
        const expiresAt = new Date();
        expiresAt.setDate(expiresAt.getDate() + 30);

        const [notification] = await this.drizzle.db
          .insert(notifications)
          .values({
            organisationId,
            userId: dto.userId,
            titleKey: dto.titleKey,
            data: dto.data || {},
            expiresAt,
          })
          .returning();

        // Emit real-time via WebSocket to target user
        this.eventsGateway.emitToUser(dto.userId, 'notification', {
          id: notification.id,
          titleKey: notification.titleKey,
          data: notification.data,
          createdAt: notification.createdAt,
        });

        return notification;
      }

      async findAll(organisationId: string, userId: string, query: QueryNotificationsDto) {
        const conditions = [
          eq(notifications.organisationId, organisationId),
          eq(notifications.userId, userId),
        ];

        if (query.isRead !== undefined) {
          conditions.push(eq(notifications.isRead, query.isRead));
        }

        return this.drizzle.db
          .select()
          .from(notifications)
          .where(and(...conditions))
          .orderBy(desc(notifications.createdAt))
          .limit(query.limit || 20)
          .offset(query.offset || 0);
      }

      async markRead(id: string, organisationId: string, userId: string) {
        const [updated] = await this.drizzle.db
          .update(notifications)
          .set({ isRead: true })
          .where(and(
            eq(notifications.id, id),
            eq(notifications.organisationId, organisationId),
            eq(notifications.userId, userId),
          ))
          .returning();
        return updated;
      }

      async markUnread(id: string, organisationId: string, userId: string) {
        const [updated] = await this.drizzle.db
          .update(notifications)
          .set({ isRead: false })
          .where(and(
            eq(notifications.id, id),
            eq(notifications.organisationId, organisationId),
            eq(notifications.userId, userId),
          ))
          .returning();
        return updated;
      }
    }
    ```

    - Inject EventsGateway to emit real-time events (D-02)
    - Auto-calculate expiresAt = now + 30 days (D-03)
    - findAll filters by organisationId AND userId for org-scoped access (D-04)
    - markRead/markUnread require ownership check via userId + organisationId

    ```typescript
    import { Controller, Get, Post, Patch, Body, Param, Query, UseGuards, Request } from '@nestjs/common';
    import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
    import { NotificationsService } from './notifications.service';
    import { CreateNotificationDto, QueryNotificationsDto } from './dto';

    @Controller('notifications')
    @UseGuards(JwtAuthGuard)
    export class NotificationsController {
      constructor(private notificationsService: NotificationsService) {}

      @Post()
      create(@Request() req, @Body() dto: CreateNotificationDto) {
        // req.user from JWT, req.organisationId from tenant middleware
        return this.notificationsService.create(req.organisationId, {
          ...dto,
          userId: dto.userId || req.user.id, // Default to current user if not specified
        });
      }

      @Get()
      findAll(@Request() req, @Query() query: QueryNotificationsDto) {
        return this.notificationsService.findAll(
          req.organisationId,
          req.user.id,
          query,
        );
      }

      @Patch(':id/read')
      markRead(@Request() req, @Param('id') id: string) {
        return this.notificationsService.markRead(id, req.organisationId, req.user.id);
      }

      @Patch(':id/unread')
      markUnread(@Request() req, @Param('id') id: string) {
        return this.notificationsService.markUnread(id, req.organisationId, req.user.id);
      }
    }
    ```

    - All routes protected by JwtAuthGuard
    - Extract organisationId from tenant middleware (req.organisationId)
    - Extract userId from JWT (req.user.id)
    - POST /notifications creates and emits WebSocket event
    - GET /notifications lists for current user in org
    - PATCH /:id/read and PATCH /:id/unread for individual control (D-03)
