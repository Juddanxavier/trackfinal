```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const PUBLIC_PATHS = ['/login', '/api/auth/', '/auth/', '/_next/', '/favicon.ico'];
const PROTECTED_PATHS = ['/dashboard', '/shipments', '/quotes', '/notifications', '/settings'];

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Allow public paths
  if (PUBLIC_PATHS.some(path => pathname.startsWith(path))) {
    return NextResponse.next();
  }

  // Allow root path
  if (pathname === '/') {
    return NextResponse.next();
  }

  // Check if protected path
  const isProtected = PROTECTED_PATHS.some(path => pathname.startsWith(path));

  if (!isProtected) {
    // Unknown paths, allow but could 404 later
    return NextResponse.next();
  }

  // Protected route - validate auth via /api/auth/me
  try {
    const apiUrl = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:4000/api';
    const response = await fetch(`${apiUrl}/auth/me`, {
      method: 'GET',
      credentials: 'include',
      headers: {
        'Content-Type': 'application/json',
      },
      // Don't follow redirects - 401 will be caught
      redirect: 'manual',
    });

    if (response.status === 200) {
      // Authenticated - allow
      return NextResponse.next();
    }

    // 401 or other - redirect to login
    const loginUrl = new URL('/login', request.url);
    loginUrl.searchParams.set('redirect', pathname);
    return NextResponse.redirect(loginUrl);
  } catch {
    // Network error or other - redirect to login
    const loginUrl = new URL('/login', request.url);
    loginUrl.searchParams.set('redirect', pathname);
    return NextResponse.redirect(loginUrl);
  }
}

export const config = {
  matcher: [
    /*
     * Match all request paths except for the ones starting with:
     * - _next/static (static files)
     * - _next/image (image optimization files)
     * - favicon.ico (favicon file)
     */
    '/((?!_next/static|_next/image|favicon.ico).*)',
  ],
};
```

```typescript
'use client';

import { useAuth } from '@/hooks/use-auth';
import { Loader2 } from 'lucide-react';

export default function ProtectedLayout({ children }: { children: React.ReactNode }) {
  const { isLoading, isAuthenticated } = useAuth();

  if (isLoading) {
    return (
      <div className="flex min-h-screen items-center justify-center">
        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
      </div>
    );
  }

  // If somehow we get here without auth, show nothing (middleware redirects)
  if (!isAuthenticated) {
    return null;
  }

  return <>{children}</>;
}
```

Note: In Next.js 16, route groups with parentheses don't create URL paths. (protected)/dashboard is just /dashboard.
